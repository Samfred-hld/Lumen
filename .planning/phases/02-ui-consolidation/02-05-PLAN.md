  setLocal('onboarded', 'true');
  await setSettingToCloud('onboarded', 'true');
}

  const cloud = await getSettingFromCloud('onboarded');
  if (cloud) { setLocal('onboarded', cloud); return cloud; }
  return isOnboarded();
}
